💻 Quick Start Code

Get started with RetryFlow in just a few lines.

import { retry } from 'retryflow';

// Basic usage - retry a flaky API call
const data = await retry(async () => {
  const res = await fetch('https://api.example.com/data');
  if (!res.ok) throw new Error(`HTTP ${res.status}`);
  return res.json();
}, {
  retries: 3,
  delay: 1000,
  backoff: 'exponential',
  factor: 2,
  timeout: 5000,
  onRetry: ({ attempt, error, delay }) => {
    console.log(`Retry #${attempt} in ${delay}ms: ${error.message}`);
  }
});
